const SPREADSHEET_ID = "1YH2m_YXAoe-23x0_hOzp4vBIircb0tyznv9z_9_KOqw"; const TIME_ZONE = "Asia/Manila"; // MAIN WEB APP ROUTER function doGet(e) { try { const params = e && e.parameter ? e.parameter : {}; const action = params.action || ""; if (action === "generateQR") { return generateQR(params.studentNumber, params.type); } if (action === "scan") { return scanQRCode(params.token); } // ADMIN DASHBOARD if (action === "dashboard") { return jsonResponse(getDashboardSummary()); } return jsonResponse({ success: true, system: "PLMunTrack", message: "Google Sheets connected successfully!" }); } catch (error) { return jsonResponse({ success: false, message: error.message }); } } // HANDLE WEBSITE REGISTRATION function doPost(e) { try { const data = JSON.parse(e.postData.contents || "{}"); if (data.action === "registerStudent") { return registerStudent(data); } return jsonResponse({ success: false, message: "Invalid action." }); } catch (error) { return jsonResponse({ success: false, message: error.message }); } } // REGISTER STUDENT function registerStudent(data) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const sheet = ss.getSheetByName("People"); if (!sheet) throw new Error("People sheet not found."); const studentNumber = String(data.studentNumber || "").trim(); const fullName = String(data.fullName || "").trim(); const email = String(data.email || "").trim(); const course = String(data.course || "").trim(); const yearSection = String(data.yearSection || "").trim(); if (!studentNumber || !fullName || !email) { throw new Error( "Student number, full name, and email are required." ); } const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const lastRow = sheet.getLastRow(); if (lastRow >= 2) { const numbers = sheet .getRange(2, 1, lastRow - 1, 1) .getDisplayValues() .flat() .map(value => value.trim()); if (numbers.includes(studentNumber)) { throw new Error("Student number is already registered."); } } sheet.appendRow([ studentNumber, "STUDENT", fullName, email, course, yearSection, "Active" ]); } finally { lock.releaseLock(); } return jsonResponse({ success: true, message: "Student registered successfully!" }); } // GENERATE DAILY QR TOKEN function generateQR(studentNumber, type) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const people = ss.getSheetByName("People"); const tokens = ss.getSheetByName("QR_Tokens"); if (!people || !tokens) { throw new Error("People or QR_Tokens sheet not found."); } studentNumber = String(studentNumber || "").trim(); type = String(type || "").trim().toUpperCase(); if (!studentNumber) { throw new Error("Student number is required."); } if (!["IN", "OUT"].includes(type)) { throw new Error("QR type must be IN or OUT."); } const today = Utilities.formatDate( new Date(), TIME_ZONE, "yyyy-MM-dd" ); const peopleData = people.getDataRange().getDisplayValues(); const person = peopleData.slice(1).find(row => String(row[0]).trim() === studentNumber && String(row[6]).trim().toLowerCase() === "active" ); if (!person) { throw new Error( "Active student not found. Register the student first." ); } const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const tokenData = tokens.getDataRange().getDisplayValues(); // Reuse today's active token for the same student and QR type. const existing = tokenData.slice(1).find(row => String(row[1]).trim() === studentNumber && String(row[3]).trim() === today && String(row[5]).trim().toLowerCase() === "active" && String(row[6]).trim().toUpperCase() === type ); if (existing) { return jsonResponse({ success: true, studentNumber: studentNumber, fullName: person[2], type: type, date: today, token: existing[2], message: "Today's QR token already exists." }); } // Random token only; no personal information in the QR. const token = Utilities.getUuid() + "-" + Utilities.getUuid(); const tokenId = "QR-" + Utilities.getUuid(); tokens.appendRow([ tokenId, studentNumber, token, today, new Date(), "Active", type ]); return jsonResponse({ success: true, studentNumber: studentNumber, fullName: person[2], type: type, date: today, token: token, message: "Daily QR token generated successfully." }); } finally { lock.releaseLock(); } } // VALIDATE QR TOKEN AND RECORD ATTENDANCE function scanQRCode(qrToken) { qrToken = String(qrToken || "").trim(); if (!qrToken) { throw new Error("QR token is required."); } const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const tokens = ss.getSheetByName("QR_Tokens"); const people = ss.getSheetByName("People"); const attendance = ss.getSheetByName("Attendance"); if (!tokens || !people || !attendance) { throw new Error( "Required sheet missing: QR_Tokens, People, or Attendance." ); } const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const now = new Date(); const today = Utilities.formatDate( now, TIME_ZONE, "yyyy-MM-dd" ); const currentTime = Utilities.formatDate( now, TIME_ZONE, "HH:mm:ss" ); const tokenData = tokens.getDataRange().getDisplayValues(); const tokenIndex = tokenData.findIndex((row, index) => index > 0 && String(row[2]).trim() === qrToken ); if (tokenIndex === -1) { throw new Error("Invalid QR token."); } const tokenRow = tokenData[tokenIndex]; const tokenSheetRow = tokenIndex + 1; if (String(tokenRow[5]).trim().toLowerCase() !== "active") { throw new Error("This QR token is not active."); } if (String(tokenRow[3]).trim() !== today) { tokens.getRange(tokenSheetRow, 6).setValue("Expired"); throw new Error("This QR code has expired. Generate today's QR."); } const studentNumber = String(tokenRow[1]).trim(); const type = String(tokenRow[6]).trim().toUpperCase(); if (!["IN", "OUT"].includes(type)) { throw new Error("Invalid QR type."); } const peopleData = people.getDataRange().getDisplayValues(); const person = peopleData.slice(1).find(row => String(row[0]).trim() === studentNumber && String(row[6]).trim().toLowerCase() === "active" ); if (!person) { throw new Error("Student is not registered or is inactive."); } const attendanceData = attendance.getDataRange().getDisplayValues(); const attendanceIndex = attendanceData.findIndex((row, index) => index > 0 && String(row[1]).trim() === studentNumber && String(row[3]).trim() === today ); let attendanceRowNumber; let message; if (type === "IN") { if (attendanceIndex !== -1) { const existingAttendance = attendanceData[attendanceIndex]; if (String(existingAttendance[4]).trim()) { throw new Error("Time-In has already been recorded today."); } } const attendanceId = "ATT-" + Utilities.getUuid(); attendance.appendRow([ attendanceId, studentNumber, person[1], today, currentTime, "", "Present" ]); attendanceRowNumber = attendance.getLastRow(); message = "Time-In recorded successfully."; } else { if (attendanceIndex === -1) { throw new Error( "No Time-In found for today. Record Time-In first." ); } attendanceRowNumber = attendanceIndex + 1; const existingAttendance = attendanceData[attendanceIndex]; if (!String(existingAttendance[4]).trim()) { throw new Error("Record Time-In before Time-Out."); } if (String(existingAttendance[5]).trim()) { throw new Error("Time-Out has already been recorded today."); } attendance.getRange(attendanceRowNumber, 6) .setValue(currentTime); attendance.getRange(attendanceRowNumber, 7) .setValue("Completed"); message = "Time-Out recorded successfully."; } const audit = ss.getSheetByName("Audit"); if (audit) { audit.appendRow([ "AUD-" + Utilities.getUuid(), person[3], "ATTENDANCE_" + type, studentNumber + " - " + message, now ]); } return jsonResponse({ success: true, message: message, fullName: person[2], studentNumber: studentNumber, personType: person[1], date: today, time: currentTime, status: type === "IN" ? "Time-In" : "Time-Out" }); } finally { lock.releaseLock(); } } // ADMIN DASHBOARD SUMMARY function getDashboardSummary() { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const peopleSheet = ss.getSheetByName("People"); const attendanceSheet = ss.getSheetByName("Attendance"); if (!peopleSheet || !attendanceSheet) { return { success: false, message: "People or Attendance sheet was not found." }; } const peopleData = peopleSheet.getDataRange().getDisplayValues(); const attendanceData = attendanceSheet.getDataRange().getDisplayValues(); if (peopleData.length === 0 || attendanceData.length === 0) { return { success: false, message: "The People or Attendance sheet has no header row." }; } const peopleHeaders = peopleData[0].map(value => String(value).trim() ); const attendanceHeaders = attendanceData[0].map(value => String(value).trim() ); const peopleStatusIndex = peopleHeaders.indexOf("Status"); const timeInIndex = attendanceHeaders.indexOf("Time_In"); const timeOutIndex = attendanceHeaders.indexOf("Time_Out"); const dateIndex = attendanceHeaders.indexOf("Date"); if ( timeInIndex === -1 || timeOutIndex === -1 || dateIndex === -1 ) { return { success: false, message: "Attendance headers must include Date, Time_In, and Time_Out." }; } const today = Utilities.formatDate( new Date(), TIME_ZONE, "yyyy-MM-dd" ); const totalPeople = peopleData.slice(1).filter(row => { if (peopleStatusIndex === -1) return true; return String(row[peopleStatusIndex]) .trim() .toLowerCase() === "active"; }).length; let totalIn = 0; let totalOut = 0; let inside = 0; attendanceData.slice(1).forEach(row => { const rowDate = normalizeSheetDate(row[dateIndex]); if (rowDate !== today) return; const timeIn = String(row[timeInIndex] || "").trim(); const timeOut = String(row[timeOutIndex] || "").trim(); if (timeIn !== "") { totalIn++; if (timeOut === "") { inside++; } else { totalOut++; } } }); return { success: true, date: today, totalPeople: totalPeople, totalIn: totalIn, totalOut: totalOut, inside: inside }; } // NORMALIZE DATE FROM GOOGLE SHEETS function normalizeSheetDate(value) { if (value instanceof Date && !isNaN(value.getTime())) { return Utilities.formatDate(value, TIME_ZONE, "yyyy-MM-dd"); } const text = String(value || "").trim(); if (/^\d{4}-\d{2}-\d{2}$/.test(text)) { return text; } // Handle dates displayed as MM/DD/YYYY. const match = text.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/); if (match) { const month = match[1].padStart(2, "0"); const day = match[2].padStart(2, "0"); return match[3] + "-" + month + "-" + day; } return text.slice(0, 10); } // JSON RESPONSE function jsonResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
